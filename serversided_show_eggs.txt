local LocalPlayer = game:GetService("Players").LocalPlayer
local Pets = workspace:WaitForChild("__REMOTES"):WaitForChild("Game"):WaitForChild("Pets")

-- CONFIG
local EGG     = "Snake"           -- egg model shown; any Library.Directory.Eggs name
local COPIES  = 1                 -- pets per hatch (1..3 - the reveal only slots three)
local ROUNDS  = 10                 -- how many times to fire
local SPACING = 0               -- seconds between rounds

local GOLDEN      = true         -- <<< TOGGLE: gold variant
local RAINBOW     = true         -- <<< TOGGLE: rainbow variant
local DARK_MATTER = true          -- <<< TOGGLE: dark matter variant
-- More than one on? Label keeps highest priority: Gold > Rainbow > Dark Matter

local PET_ID = "1001"             -- Library.Directory.Pets id (1001 = Cat)
local LEVEL  = 180000000000       -- 180 billion

local PET = { cn = "Cat", id = 48291037, e = false, l = LEVEL, xp = 0, n = PET_ID }
if GOLDEN      then PET.g  = true end   -- "Gold " name prefix + rare sting
if RAINBOW     then PET.r  = true end   -- "Rainbow " name prefix + rare sting
if DARK_MATTER then PET.dm = true end   -- "Dark Matter " name prefix + rare sting

local function batch()
    local b = {}
    for _ = 1, math.max(1, math.min(3, COPIES)) do
        b[#b + 1] = PET          -- push the PET itself, not an array of pets
    end
    return b
end

local function fire()
    Pets:FireServer({ { "Open", "Egg", EGG, batch() } })
end

for i = 1, ROUNDS do
    fire()
    if i < ROUNDS then task.wait(SPACING) end
end

print(("[hatch_broadcast] fired %d round(s) of %d x %s (gold=%s rainbow=%s dm=%s) lvl=%d"):format(
    ROUNDS, COPIES, PET_ID, tostring(GOLDEN), tostring(RAINBOW), tostring(DARK_MATTER), LEVEL))

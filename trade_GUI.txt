-- Variables
local selectedPlayer = nil
local playerButtons = {}

-- Functions
local function updatePlayerList()
    -- Clear existing buttons
    for _, button in pairs(playerButtons) do
        button:Destroy()
    end
    playerButtons = {}
    
    -- Create buttons for each player
    for _, player in pairs(Players:GetPlayers()) do
        if player ~= localplayer then
            local button = Instance.new("TextButton")
            button.Name = player.Name
            button.Size = UDim2.new(1, 0, 0, 30)
            button.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
            button.BackgroundTransparency = 0.3
            button.Text = player.Name .. " (Level: ?)"
            button.TextColor3 = Color3.fromRGB(255, 255, 255)
            button.TextXAlignment = Enum.TextXAlignment.Left
            button.Font = Enum.Font.Gotham
            button.TextSize = 14
            button.Parent = playerScrollingFrame
            
            local buttonCorner = Instance.new("UICorner")
            buttonCorner.CornerRadius = UDim.new(0, 4)
            buttonCorner.Parent = button
            
            -- Hover effect
            button.MouseEnter:Connect(function()
                if selectedPlayer ~= player then
                    button.BackgroundColor3 = Color3.fromRGB(55, 55, 65)
                end
            end)
            
            button.MouseLeave:Connect(function()
                if selectedPlayer ~= player then
                    button.BackgroundColor3 = Color3.fromRGB(45, 45, 55)
                end
            end)
            
            -- Selection
            button.MouseButton1Click:Connect(function()
                -- Reset previous selection
                if selectedPlayer and playerButtons[selectedPlayer] then
                    playerButtons[selectedPlayer].BackgroundColor3 = Color3.fromRGB(45, 45, 55)
                end
                
                -- Select new player
                selectedPlayer = player
                button.BackgroundColor3 = Color3.fromRGB(50, 150, 255)
                tradeButton.Text = "Trade " .. player.Name
            end)
            
            playerButtons[player] = button
        end
    end
end

local function showNotification(message, isError)
    local notification = Instance.new("Frame")
    notification.Size = UDim2.new(0, 300, 0, 50)
    notification.Position = UDim2.new(0.5, -150, 0, 50)
    notification.BackgroundColor3 = isError and Color3.fromRGB(255, 80, 80) or Color3.fromRGB(50, 200, 100)
    notification.BackgroundTransparency = 0.2
    notification.Parent = screenGui
    
    local notifCorner = Instance.new("UICorner")
    notifCorner.CornerRadius = UDim.new(0, 8)
    notifCorner.Parent = notification
    
    local notifText = Instance.new("TextLabel")
    notifText.Size = UDim2.new(1, -20, 1, 0)
    notifText.Position = UDim2.new(0, 10, 0, 0)
    notifText.BackgroundTransparency = 1
    notifText.Text = message
    notifText.TextColor3 = Color3.fromRGB(255, 255, 255)
    notifText.TextWrapped = true
    notifText.Font = Enum.Font.Gotham
    notifText.TextSize = 14
    notifText.Parent = notification
    
    -- Animate and destroy
    local tweenInfo = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.Out)
    local goal = {Position = UDim2.new(0.5, -150, 0, 50)}
    local fadeOut = TweenInfo.new(0.3, Enum.EasingStyle.Quad, Enum.EasingDirection.In)
    local fadeGoal = {Position = UDim2.new(0.5, -150, 0, -50)}
    
    task.wait(2)
    local fadeTween = TweenService:Create(notification, fadeOut, fadeGoal)
    fadeTween:Play()
    fadeTween.Completed:Connect(function()
        notification:Destroy()
    end)
end

-- Connect to trade updates
game:FindFirstChild('Trade Update', true).OnClientEvent:Connect(function(id)
    lastUpdateId = id
    tradeIdLabel.Text = "Current Trade ID: " .. tostring(id)
    tradeStatusLabel.Text = "Status: Trade pending"
    showNotification("New trade request received!", false)
end)

-- Button functionality
tradeButton.MouseButton1Click:Connect(function()
    if selectedPlayer then
        task.spawn(function()
            repeat 
                task.wait(.25) 
            until tradingRemote:InvokeServer("InvSend", selectedPlayer) == true
        end)
        showNotification("Trade request sent to " .. selectedPlayer.Name, false)
    else
        showNotification("Please select a player first!", true)
    end
end)

acceptButton.MouseButton1Click:Connect(function()
    if lastUpdateId then
        local success, err = pcall(function()
            tradingRemote:InvokeServer("Ready", lastUpdateId)
        end)
        if success then 
            showNotification("Trade accepted: " .. lastUpdateId, false)
            tradeStatusLabel.Text = "Status: Trade accepted"
        else
            showNotification("Failed to accept trade!", true)
        end
    else
        showNotification("No active trade to accept!", true)
    end
end)

cancelButton.MouseButton1Click:Connect(function()
    if lastUpdateId then
        local success, err = pcall(function()
            tradingRemote:InvokeServer("Cancel", lastUpdateId)
        end)
        if success then 
            showNotification("Trade cancelled", false)
            tradeIdLabel.Text = "Current Trade ID: None"
            tradeStatusLabel.Text = "Status: No active trade"
            lastUpdateId = nil
        else
            showNotification("Failed to cancel trade!", true)
        end
    else
        showNotification("No active trade to cancel!", true)
    end
end)

-- Toggle GUI
toggleButton.MouseButton1Click:Connect(function()
    mainFrame.Visible = not mainFrame.Visible
    if mainFrame.Visible then
        toggleButton.Text = "❌"
        updatePlayerList()
    else
        toggleButton.Text = "💰"
    end
end)

closeButton.MouseButton1Click:Connect(function()
    mainFrame.Visible = false
    toggleButton.Text = "💰"
end)

-- Update player list when players join/leave
Players.PlayerAdded:Connect(updatePlayerList)
Players.PlayerRemoving:Connect(updatePlayerList)

-- Initial update
updatePlayerList()

-- Keyboard shortcut (LeftAlt + T)
UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if not gameProcessed and input.KeyCode == Enum.KeyCode.T and UserInputService:IsKeyDown(Enum.KeyCode.LeftAlt) then
        mainFrame.Visible = not mainFrame.Visible
        if mainFrame.Visible then
            toggleButton.Text = "❌"
            updatePlayerList()
        else
            toggleButton.Text = "💰"
        end
    end
end)

print("Trading GUI loaded! Click the 💰 button or press LeftAlt + T to open")
